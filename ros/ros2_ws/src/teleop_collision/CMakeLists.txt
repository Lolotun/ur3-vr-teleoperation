cmake_minimum_required(VERSION 3.8)
project(teleop_collision)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_ros_planning REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(shape_msgs REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(collision_guard src/collision_guard.cpp)
target_compile_features(collision_guard PUBLIC cxx_std_17)
target_compile_options(collision_guard PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(collision_guard rclcpp moveit_core moveit_ros_planning moveit_msgs shape_msgs std_msgs)
install(TARGETS collision_guard DESTINATION lib/${PROJECT_NAME})
include(CTest)
if(BUILD_TESTING)
  add_executable(clearance_policy_test test/clearance_policy_test.cpp)
  target_compile_features(clearance_policy_test PRIVATE cxx_std_17)
  target_include_directories(clearance_policy_test PRIVATE src)
  add_test(NAME clearance_policy COMMAND clearance_policy_test)
endif()
ament_package()
